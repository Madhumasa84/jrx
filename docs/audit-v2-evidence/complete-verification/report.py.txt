import json,pathlib,xml.etree.ElementTree as ET
root=pathlib.Path(__file__).parent
expected=['source-suite','wheel311-suite','wheel312-suite','ruff','format','mypy','bandit','helm-lint','helm-render','make-static','wheel312-static','build','install-wheel311','install-wheel312','install-sdist311','wheel311-cli','wheel312-cli','sdist311-cli','docker-build','docker-smoke-retry','wheel311-pip-audit']
records=[]
for label in expected:
 p=root/(label+'.json')
 records.append(json.loads(p.read_text()) if p.exists() else {'name':label,'exit_code':None})
tests=[]
for label in ['source','wheel311','wheel312']:
 p=root/(label+'-tests.xml')
 if not p.exists():continue
 cases=ET.parse(p).getroot().findall('.//testcase')
 skipped=[{'name':c.attrib.get('classname','')+'.'+c.attrib['name'],'reason':c.find('skipped').attrib.get('message','')} for c in cases if c.find('skipped') is not None]
 failures=[c.attrib['name'] for c in cases if c.find('failure') is not None or c.find('error') is not None]
 tests.append({'environment':label,'total':len(cases),'passed':len(cases)-len(skipped)-len(failures),'skipped':skipped,'failures':failures})
summary={'commit':'9107b0db87676195e21f49c2de5862d8f5657fbb','checks':records,'tests':tests,'all_expected_pass':all(r['exit_code']==0 for r in records)}
(root/'results.json').write_text(json.dumps(summary,indent=2))
lines=['# Complete verification of JRX commit 9107b0d','','Repository code was unchanged during testing. The remote branch SHA was verified before testing.','', '## Full suites','','| Environment | Passed | Skipped | Failed |','| --- | ---: | ---: | ---: |']
for t in tests:lines.append(f"| {t['environment']} | {t['passed']} | {len(t['skipped'])} | {len(t['failures'])} |")
lines+=['','Each full suite includes broker/TLS, MCP, approval, session, audit, concurrency, intent, authority, harness and disposable Docker integration tests. Installed-wheel tests removed PYTHONPATH and used importlib mode. Module origins and dependency versions are recorded separately.','','## Commands','','| Check | Exit | Seconds |','| --- | ---: | ---: |']
for r in records:lines.append(f"| [{r['name']}]({r['name']}.log) | {r['exit_code'] if r['exit_code'] is not None else 'pending'} | {r.get('elapsed_seconds','')} |")
lines+=['','Exact commands, working directories, results and durations are in results.json; raw transcripts and JUnit XML are retained alongside this report.','','## Scope and limitations','','- Only the paid live TypeSafe integration test is intentionally skipped. Synthetic data, mocks and local certificates are used. No paid API calls, external deployments or release publication.','- 28 CLI checks run independently against wheel Python 3.11, wheel Python 3.12 and sdist Python 3.11, including audit verify/tail, session stop/status, unknown MCP denial without forwarding, benchmarks, and intent/authority lifecycle.','- Bandit reports zero medium/high findings under the requested -ll threshold; 49 low findings remain.','- pip-audit checks the fresh installed dependencies; the unpublished jrx distribution has no PyPI advisory record and is reported separately as unauditable by that service.','- The first Docker smoke driver exceeded its 120-second external limit during concurrent verification. That failure is retained in docker-smoke.log/json. A later serial retry uses a longer driver limit; JRX security timeouts and enforcement are unchanged.','- macOS, Windows, ARM and paid live evaluator behavior are not tested by this Linux run. Passing suites do not establish perfect command interpretation, runtime confinement, host isolation or rollback resistance.','- Original user checkout edits and the hardening branch are preserved; the validation clone contains no source changes.']
(root/'VERIFICATION.md').write_text('\n'.join(lines)+'\n')
print(json.dumps({'tests':tests,'checks_finished':sum(r['exit_code'] is not None for r in records),'checks_expected':len(records),'all_expected_pass':summary['all_expected_pass']}))
