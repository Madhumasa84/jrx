def suite(label, xml_name, commit, scanner):
    cases = ET.parse(root/xml_name).getroot().findall('.//testcase')
    skipped = [{'test':case.attrib.get('classname', '') + '.' + case.attrib['name'],
                'reason':case.find('skipped').get('message', '')}
               for case in cases if case.find('skipped') is not None]
    failed = [case.attrib.get('classname', '') + '.' + case.attrib['name']
              for case in cases if case.find('failure') is not None or case.find('error') is not None]
    log = (root/(label+'.log')).read_text()
    duration = re.search(r'\d+ passed.*? in ([\d.]+)s', log)
    return {'environment':label, 'code_sha':commit, 'scanner':scanner,
            'total':len(cases), 'passed':len(cases)-len(skipped)-len(failed),
            'skipped':skipped, 'failed':failed,
            'pytest_seconds':float(duration.group(1)) if duration else None}
